💡 EJEMPLO DE IMPLEMENTACIÓN COMPLETA:

```python
# 1. Cargar divisiones
import pandas as pd
from sklearn.ensemble import IsolationForest
from sklearn.metrics import classification_report

# Datos para modelo de anomalías
X_normal = pd.read_csv('C:\Users\francisco.lopez\KIKO_TFG\Working_Directory\Resultados/specialized_nsl_kdd_processed_X_D1_normal_for_anomaly.csv')

# Datos para evaluación
X_test = pd.read_csv('C:\Users\francisco.lopez\KIKO_TFG\Working_Directory\Resultados/specialized_nsl_kdd_processed_X_D2_complete_test.csv')
y_test = pd.read_csv('C:\Users\francisco.lopez\KIKO_TFG\Working_Directory\Resultados/specialized_nsl_kdd_processed_y_category_D2_complete_test.csv')

# Datos para firmas
X_attacks = pd.read_csv('C:\Users\francisco.lopez\KIKO_TFG\Working_Directory\Resultados/specialized_nsl_kdd_processed_X_D3_known_attacks_for_signatures.csv')
y_attacks = pd.read_csv('C:\Users\francisco.lopez\KIKO_TFG\Working_Directory\Resultados/specialized_nsl_kdd_processed_y_category_D3_known_attacks_for_signatures.csv')

# 2. Entrenar modelo de anomalías
anomaly_model = IsolationForest(contamination=0.1, random_state=42)
anomaly_model.fit(X_normal)  # Solo con datos normales

# 3. Extraer firmas de ataques
# Análisis estadístico por categoría
for category in y_attacks['category_original'].unique():
    mask = y_attacks['category_original'] == category
    category_data = X_attacks[mask]
    # Análizar características distintivas
    print(f'Patrones para {category}: {category_data.describe()}')

# 4. Evaluar en datos de prueba
anomaly_predictions = anomaly_model.predict(X_test)
# -1 = anomalía, 1 = normal
print('Evaluación del modelo de anomalías:')
# Convertir a binario para evaluación
y_binary = (y_test['category_original'] != 'normal').astype(int)
anomaly_binary = (anomaly_predictions == -1).astype(int)
print(classification_report(y_binary, anomaly_binary))
```
